String processor(const String& var) {
    if (var == "DELAY_VAL") return String(nextDelay);
    if (var == "LAST_IP") return lastClientIP;
    if (var == "LAST_DATA") return lastReceivedData;
    if (var == "STA_IP") return WiFi.localIP().toString();
    if (var == "AP_IP") return WiFi.softAPIP().toString();
    return String();
}

void setup() {
    Serial.begin(115200);
    preferences.begin("settings", false);
    nextDelay = preferences.getInt("delay", 5000);

    WiFi.mode(WIFI_AP_STA);
    server.startAP(); 

    WiFi.begin(local_ssid);
    Serial.print("Connecting to Wi-Fi...");
    unsigned long start = millis();
    while (WiFi.status() != WL_CONNECTED && millis() - start < 10000) { delay(500); Serial.print("."); }

    Serial.println("\n--- Networks Ready ---");

    Serial.print("Local IP (Connect here for Web UI): ");

    Serial.println(WiFi.localIP());

    Serial.print("AP IP (Clients connect here): ");

    Serial.println(WiFi.softAPIP());

    // Routes
    webServer.on("/", HTTP_GET, [](AsyncWebServerRequest *request){
        request->send_P(200, "text/html", index_html, processor);
    });

    // New JSON endpoint for AJAX auto-refresh
    webServer.on("/status", HTTP_GET, [](AsyncWebServerRequest *request){
        String json = "{\"lastData\":\"" + lastReceivedData + "\", \"lastIP\":\"" + lastClientIP + "\"}";
        request->send(200, "application/json", json);
    });

    webServer.on("/update", HTTP_GET, [](AsyncWebServerRequest *request){
        if (request->hasParam("delay")) {
            nextDelay = request->getParam("delay")->value().toInt();
            preferences.putInt("delay", nextDelay);
        }
        request->redirect("/");
    });

    webServer.begin();
}

void loop() {
    if (server.acceptClient()) Serial.println("Client Connected");

    Opcode code;
    String data;
    String remoteIP;

    if (server.receivePacket(code, data, remoteIP)) {
        lastReceivedData = data;
        lastClientIP = remoteIP;
        server.sendUpdate(nextDelay);
    }
}
